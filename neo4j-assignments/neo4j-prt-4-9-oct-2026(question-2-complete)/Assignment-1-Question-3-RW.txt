e.] Works_in relationships


MATCH (e1:Employee {emp_id:101}), (e2:Employee {emp_id:102}),
      (e3:Employee {emp_id:103}), (e4:Employee {emp_id:104}),
      (e5:Employee {emp_id:105}),
      (d1:Department {name:"Computer Science"}),
      (d2:Department {name:"Human Resources"})
CREATE
(e1)-[:Works_in]->(d1),
(e2)-[:Works_in]->(d1),
(e3)-[:Works_in]->(d1),
(e4)-[:Works_in]->(d2),
(e5)-[:Works_in]->(d2);

---

x.] Return All Nodes And Relationships That Exist


MATCH (n)
OPTIONAL MATCH (n)-[r]-()
RETURN n, r;

OR

MATCH (n)
OPTIONAL MATCH (n)-[r]-(m)
RETURN n, r, m;

Queries:


// 1.] List the names of employees in department "BPO"

MATCH (e:Employee)-[:Works_in]->(d:Department)
WHERE d.name = "BPO"
RETURN e.name AS Employee;

// 2.] List the projects along with their properties,
//     controlled by department "IT"

MATCH (p:Projects)-[:Controlled_by]->(d:Department)
WHERE d.name = "IT"
RETURN p;

// 3.] List the departments along with the count of employees in it

MATCH (d:Department)
OPTIONAL MATCH (e:Employee)-[:Works_in]->(d)
RETURN d.name AS Department, COUNT(e) AS Employee_Count;

// 4.] List the skillset for an employee "Jayson"

MATCH (e:Employee)-[:Has_acquired]->(s:Skillset)
WHERE e.name = "Jayson"
RETURN e.name AS Employee, s.name AS Skillset;

// 5.] List the projects controlled by department "BPO"
//     and have employees of the same department working in it

MATCH (p:Projects)-[:Controlled_by]->(d:Department)
MATCH (e:Employee)-[:Works_in]->(d)
MATCH (e)-[:Assigned_to]->(p)
WHERE d.name = "BPO"
RETURN DISTINCT p.name AS Project;

// 6.] List the names of the projects belonging to departments
//     managed by employee "Kevin"

MATCH (e:Employee)-[:Project_manager]->(p:Projects)
MATCH (p)-[:Controlled_by]->(d:Department)
WHERE e.name = "Kevin"
RETURN DISTINCT p.name AS Project;
